# Học sinh tự viết mini quiz
diem = 0

cau_hoi = [
    ("Python dùng lệnh nào để in?", "print"),
    ("Vòng lặp biết trước số lần?", "for")
]

for cau, dap_an in cau_hoi:
    tra_loi = input(cau + " ").lower()
    if tra_loi == dap_an:
        diem += 1

print("Điểm của em:", diem)
